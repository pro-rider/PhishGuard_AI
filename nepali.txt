"""
PhishGuard_AI - Hash IOC Extractor
Version: 19.0.0

Purpose
-------
Extract cryptographic hash indicators from untrusted text.

Supported digest families
-------------------------
MD5
SHA-1
SHA-224
SHA-256
SHA-384
SHA-512
SHA3-224
SHA3-256
SHA3-384
SHA3-512
BLAKE2S-256
BLAKE2B-512

Additional formats
------------------
* Explicit algorithm prefixes:
    md5:<digest>
    sha1:<digest>
    sha256:<digest>
    sha512:<digest>
    sha3-256:<digest>
    blake2b:<digest>

* SRI:
    sha256-<base64>
    sha384-<base64>
    sha512-<base64>

* Defanged hexadecimal digests:
    abc[.]def
    abc(dot)def
    abc [.] def

Security properties
-------------------
* No network access
* No filesystem access
* No subprocess execution
* Strict input-size limits
* Strict IOC-count limits
* Bounded candidate lengths
* Bounded extraction time
* ReDoS-resistant candidate discovery strategy
* Exact original-text span mapping
* Deterministic output
* Stable IOC IDs
* Defensive Unicode handling
* Context sanitization
* Optional Git/SHA context suppression
* Explicit ambiguity representation
* Provenance preservation
* Async CPU offloading

Important design rule
---------------------
Extraction is NOT reputation.

A digest being syntactically valid does not imply:
    malicious
    benign
    known-good
    trustworthy

The reputation layer must make that decision.
"""

class HashIOC(BaseModel):
    """
    Immutable extracted hash IOC.

    start/end always refer to the ORIGINAL input string.
    """

    model_config = ConfigDict(
        frozen=True,
        extra="forbid",
    )

    ioc_id: str = Field(min_length=16, max_length=128)

    kind: str = Field(default="hash", min_length=1, max_length=32)

    algorithm: HashAlgorithm

    value: str = Field(
        min_length=32,
        max_length=128,
    )

    raw_value: str = Field(
        min_length=1,
        max_length=512,
    )

    start: int = Field(ge=0)

    end: int = Field(gt=0)

    confidence: float = Field(ge=0.0, le=1.0)

    match_type: HashMatchType

    encoding: HashEncoding

    normalization_actions: tuple[NormalizationAction, ...] = ()

    alternate_algorithms: tuple[HashAlgorithm, ...] = ()

    source: str = Field(
        default="text",
        min_length=1,
        max_length=64,
    )

    context: str = Field(
        default="",
        max_length=2_000,
    )

    metadata: Mapping[str, str | int | bool | float | tuple[str, ...]] = {}

    @field_validator("value")
    @classmethod
    def validate_hex_value(cls, value: str) -> str:
        if len(value) not in {32, 40, 56, 64, 96, 128}:
            raise ValueError("unsupported digest length")

        if any(ch not in "0123456789abcdef" for ch in value):
            raise ValueError("digest must be lowercase hexadecimal")

        return value

    @field_validator("end")
    @classmethod
    def validate_span(cls, value: int, info):
        start = info.data.get("start")
        if start is not None and value <= start:
            raise ValueError("end must be greater than start")
        return value

# Metrics

class HashExtractorMetrics:
    """
    Low-contention process metrics.

    Counters are kept per-thread and merged on snapshot.

    This avoids a global mutex on every candidate evaluation.
    """

    def __init__(self) -> None:
        self._local = threading.local()
        self._registry_lock = threading.Lock()
        self._registries: list[dict[str, int]] = []

    def _counter(self) -> dict[str, int]:
        counters = getattr(self._local, "counters", None)

        if counters is None:
            counters = {}
            self._local.counters = counters

            with self._registry_lock:
                self._registries.append(counters)

        return counters

    def increment(self, name: str, amount: int = 1) -> None:
        counters = self._counter()
        counters[name] = counters.get(name, 0) + amount

    def snapshot(self) -> dict[str, int]:
        merged: Counter[str] = Counter()

        with self._registry_lock:
            registries = tuple(self._registries)

        for counters in registries:
            merged.update(counters)

        return dict(merged)

    def reset(self) -> None:
        with self._registry_lock:
            self._registries.clear()

        self._local.counters = {}


_GLOBAL_METRICS = HashExtractorMetrics()

class HashIOCExtractor:
    """
    Production hash IOC extractor.

    Thread-safe after construction.

    The extractor itself does not perform network operations.
    """

    def __init__(
        self,
        config: Optional[HashExtractorConfig] = None,
        *,
        metrics: Optional[HashExtractorMetrics] = None,
    ) -> None:
        self.config = config or HashExtractorConfig()
        self.metrics = metrics or _GLOBAL_METRICS

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------

    def extract(self, text: str) -> HashExtractionResult:
        started = time.monotonic()

        self.metrics.increment("extract_calls")

        if not isinstance(text, str):
            self.metrics.increment("invalid_input")
            return self._failure(
                started,
                errors=("invalid_input_type",),
            )

        if not text:
            self.metrics.increment("empty_input")
            return self._failure(
                started,
                errors=(RejectionReason.EMPTY.value,),
            )

        if len(text) > self.config.max_input_chars:
            self.metrics.increment("input_too_large")
            return self._failure(
                started,
                errors=("input_too_large",),
            )

        try:
            working, index_map = self._normalize_input(text)

            candidates = self._discover_candidates(
                working,
                index_map,
                started,
            )

            self.metrics.increment(
                "candidates_seen",
                len(candidates),
            )

            deconflicted = self._deconflict_candidates(candidates)

            iocs = self._materialize_iocs(
                text,
                working,
                deconflicted,
                started,
            )

            iocs = self._deduplicate_iocs(iocs)

            truncated = (
                len(iocs) >= self.config.max_iocs
                or len(candidates) >= self.config.max_candidates
                or self._budget_exceeded(started)
            )

            if iocs:
                self.metrics.increment(
                    "iocs_accepted",
                    len(iocs),
                )

            elapsed_ms = self._elapsed_ms(started)

            return HashExtractionResult(
                success=True,
                iocs=tuple(iocs[: self.config.max_iocs]),
                truncated=truncated,
                elapsed_ms=elapsed_ms,
                candidates_seen=len(candidates),
                extractor_errors=(),
                metrics_snapshot=self.metrics.snapshot(),
            )

        except Exception:
            self.metrics.increment("extractor_errors")

            return self._failure(
                started,
                errors=("extraction_failed",),
            )

    async def extract_async(self, text: str) -> HashExtractionResult:
        """
        Offload CPU-bound extraction from the event loop.
        """

        return await asyncio.to_thread(self.extract, text)

    def metrics_snapshot(self) -> Mapping[str, int]:
        return self.metrics.snapshot()

    # ------------------------------------------------------------------
    # Input normalization with exact offset map
    # ------------------------------------------------------------------

    def _normalize_input(
        self,
        text: str,
    ) -> tuple[str, list[int]]:
        """
        Normalize text while preserving a working-index -> original-index map.

        Each character in the working string maps to an original source
        character position.

        For NFKC expansions, all generated characters map to the originating
        source position. This guarantees spans remain anchored to real input.
        """

        output: list[str] = []
        index_map: list[int] = []

        for original_index, char in enumerate(text):
            value = char

            if self.config.remove_zero_width and char in ZERO_WIDTH_CHARS:
                self.metrics.increment("zero_width_removed")
                continue

            if self.config.unicode_nfkc:
                value = unicodedata.normalize("NFKC", value)

                if value != char:
                    self.metrics.increment("unicode_normalized")

            if not value:
                continue

            for normalized_char in value:
                output.append(normalized_char)
                index_map.append(original_index)

        return "".join(output), index_map

    # ------------------------------------------------------------------
    # Candidate discovery
    # ------------------------------------------------------------------

    def _discover_candidates(
        self,
        text: str,
        index_map: Sequence[int],
        started: float,
    ) -> list[_Candidate]:
        candidates: list[_Candidate] = []

        # Explicit prefixes have highest priority.
        if self.config.allow_explicit_prefixes:
            candidates.extend(
                self._discover_prefixed(
                    text,
                    index_map,
                    started,
                    remaining=self.config.max_candidates,
                )
            )

        if self._budget_exceeded(started):
            return candidates

        # SRI is explicit and therefore higher confidence than a bare digest.
        if self.config.allow_sri:
            remaining = self.config.max_candidates - len(candidates)

            if remaining > 0:
                candidates.extend(
                    self._discover_sri(
                        text,
                        index_map,
                        started,
                        remaining=remaining,
                    )
                )

        if self._budget_exceeded(started):
            return candidates

        remaining = self.config.max_candidates - len(candidates)

        if remaining > 0:
            candidates.extend(
                self._discover_generic(
                    text,
                    index_map,
                    started,
                    remaining=remaining,
                )
            )

        return candidates[: self.config.max_candidates]

    def _discover_prefixed(
        self,
        text: str,
        index_map: Sequence[int],
        started: float,
        *,
        remaining: int,
    ) -> list[_Candidate]:
        results: list[_Candidate] = []

        # Deliberately simple and bounded.
        #
        # The digest itself is not captured with a huge regex alternation.
        # We capture a conservative token and validate it afterwards.

        pattern = re.compile(
            r"(?i)\b"
            r"(md5|md-5|sha1|sha-1|sha224|sha-224|"
            r"sha256|sha-256|sha384|sha-384|sha512|sha-512|"
            r"sha3[-_]224|sha3[-_]256|sha3[-_]384|sha3[-_]512|"
            r"blake2s(?:-256)?|blake2b(?:-512)?)"
            r"\s*[:=]\s*"
            r"([0-9a-fA-F]{32,128})"
            r"(?![0-9a-fA-F])",
        )

        for match in pattern.finditer(text):
            if len(results) >= remaining:
                break

            if self._budget_exceeded(started):
                break

            algorithm = ALGORITHM_ALIASES.get(
                match.group(1).lower()
            )

            if algorithm is None:
                continue

            digest = match.group(2)

            if not self._valid_digest_for_algorithm(
                digest,
                algorithm,
            ):
                continue

            source_start = self._map_start(
                match.start(),
                index_map,
            )

            source_end = self._map_end(
                match.end(),
                index_map,
                source_start,
            )

            results.append(
                _Candidate(
                    start=source_start,
                    end=source_end,
                    raw_value=self._safe_slice(
                        index_map,
                        match.start(),
                        match.end(),
                        fallback=digest,
                    ),
                    explicit_algorithm=algorithm,
                    encoding=HashEncoding.HEX,
                    match_type=HashMatchType.EXACT,
                    priority=100,
                )
            )

            self.metrics.increment("prefixed_candidates")

        return results

    def _discover_sri(
        self,
        text: str,
        index_map: Sequence[int],
        started: float,
        *,
        remaining: int,
    ) -> list[_Candidate]:
        results: list[_Candidate] = []

        pattern = re.compile(
            r"(?i)\b"
            r"(sha256|sha384|sha512)"
            r"-"
            r"([A-Za-z0-9+/]{20,100}={0,2})"
            r"(?![A-Za-z0-9+/=])",
        )

        sri_algorithms = {
            "sha256": HashAlgorithm.SHA256,
            "sha384": HashAlgorithm.SHA384,
            "sha512": HashAlgorithm.SHA512,
        }

        for match in pattern.finditer(text):
            if len(results) >= remaining:
                break

            if self._budget_exceeded(started):
                break

            algorithm = sri_algorithms[match.group(1).lower()]

            try:
                decoded = base64.b64decode(
                    match.group(2),
                    validate=True,
                )
            except (binascii.Error, ValueError):
                self.metrics.increment("invalid_sri")
                continue

            expected_bytes = {
                HashAlgorithm.SHA256: 32,
                HashAlgorithm.SHA384: 48,
                HashAlgorithm.SHA512: 64,
            }[algorithm]

            if len(decoded) != expected_bytes:
                self.metrics.increment("invalid_sri")
                continue

            digest = decoded.hex()

            source_start = self._map_start(
                match.start(),
                index_map,
            )

            source_end = self._map_end(
                match.end(),
                index_map,
                source_start,
            )

            results.append(
                _Candidate(
                    start=source_start,
                    end=source_end,
                    raw_value=self._safe_slice(
                        index_map,
                        match.start(),
                        match.end(),
                        fallback=match.group(0),
                    ),
                    explicit_algorithm=algorithm,
                    encoding=HashEncoding.SRI_BASE64,
                    match_type=HashMatchType.SRI,
                    priority=120,
                )
            )

            # Digest is validated by the byte length above.
            _ = digest

            self.metrics.increment("sri_candidates")

        return results

    def _discover_generic(
        self,
        text: str,
        index_map: Sequence[int],
        started: float,
        *,
        remaining: int,
    ) -> list[_Candidate]:
        results: list[_Candidate] = []

        """
        Generic discovery intentionally scans hexadecimal runs rather than
        using a collection of nested regexes.

        This makes the complexity much more predictable for hostile input.
        """

        run_start: Optional[int] = None

        for index, char in enumerate(text):
            if self._budget_exceeded(started):
                self.metrics.increment("time_budget_exceeded")
                break

            is_hex = char in "0123456789abcdefABCDEF"

            if is_hex:
                if run_start is None:
                    run_start = index
                continue

            if run_start is not None:
                self._consume_hex_run(
                    text,
                    index_map,
                    run_start,
                    index,
                    results,
                    remaining,
                )

                if len(results) >= remaining:
                    break

                run_start = None

        if (
            run_start is not None
            and len(results) < remaining
            and not self._budget_exceeded(started)
        ):
            self._consume_hex_run(
                text,
                index_map,
                run_start,
                len(text),
                results,
                remaining,
            )

        # Defanged candidates are handled separately.
        if (
            self.config.allow_defanged
            and len(results) < remaining
            and not self._budget_exceeded(started)
        ):
            self._discover_defanged(
                text,
                index_map,
                started,
                results,
                remaining - len(results),
            )

        return results

    def _consume_hex_run(
        self,
        text: str,
        index_map: Sequence[int],
        start: int,
        end: int,
        results: list[_Candidate],
        remaining: int,
    ) -> None:
        length = end - start

        if length < DEFAULT_MIN_DIGEST_LENGTH:
            return

        if length > DEFAULT_MAX_DIGEST_LENGTH:
            # Avoid producing every possible 32/40/56/... substring from a
            # huge hex blob. Search only structurally separated sub-runs.
            return

        if not self._valid_hex(text[start:end]):
            return

        if self._looks_like_non_hash_token(
            text,
            start,
            end,
        ):
            return

        digest = text[start:end].lower()

        if len(results) >= remaining:
            return

        source_start = self._map_start(
            start,
            index_map,
        )

        source_end = self._map_end(
            end,
            index_map,
            source_start,
        )

        algorithms = PREFERRED_ALGORITHMS_BY_LENGTH.get(
            len(digest),
            (),
        )

        if not algorithms:
            return

        explicit = algorithms[0]

        match_type = (
            HashMatchType.AMBIGUOUS
            if len(algorithms) > 1
            else HashMatchType.EXACT
        )

        if (
            self.config.reject_git_commit_patterns
            and self._looks_like_git_context(
                text,
                start,
                end,
            )
        ):
            self.metrics.increment("git_context_rejected")
            return

        results.append(
            _Candidate(
                start=source_start,
                end=source_end,
                raw_value=text[start:end],
                explicit_algorithm=None,
                encoding=HashEncoding.HEX,
                match_type=match_type,
                priority=50 + len(digest),
            )
        )

        self.metrics.increment("generic_candidates")

    # ------------------------------------------------------------------
    # Defanged discovery
    # ------------------------------------------------------------------

    def _discover_defanged(
        self,
        text: str,
        index_map: Sequence[int],
        started: float,
        results: list[_Candidate],
        remaining: int,
    ) -> None:
        """
        Conservative defanged scanner.

        We do not use a large nested regex.

        Instead, identify regions containing hexadecimal material plus
        known separator tokens.
        """

        markers = (
            "[.]",
            "(.)",
            "{.}",
            "[dot]",
            "(dot)",
            "{dot}",
            "[dot.]",
            "[.]",
        )

        marker_positions: list[tuple[int, int]] = []

        lowered = text.lower()

        for marker in markers:
            if self._budget_exceeded(started):
                return

            offset = 0

            while True:
                position = lowered.find(marker, offset)

                if position < 0:
                    break

                marker_positions.append(
                    (position, position + len(marker))
                )

                offset = position + len(marker)

                if len(marker_positions) >= self.config.max_candidates:
                    return

        if not marker_positions:
            return

        marker_positions.sort()

        for marker_start, marker_end in marker_positions:
            if len(results) >= remaining:
                return

            if self._budget_exceeded(started):
                return

            left = marker_start
            right = marker_end

            # Expand around the marker only through hex/space/defang markers.
            left_count = 0

            while left > 0 and left_count < 256:
                previous = lowered[left - 1]

                if previous in "0123456789abcdef":
                    left -= 1
                    left_count += 1
                    continue

                break

            right_count = 0

            while right < len(lowered) and right_count < 256:
                current = lowered[right]

                if current in "0123456789abcdef":
                    right += 1
                    right_count += 1
                    continue

                break

            if right <= left:
                continue

            raw_region = text[left:right]

            normalized = self._normalize_defanged_digest(
                raw_region,
            )

            if normalized is None:
                continue

            digest, actions = normalized

            if len(digest) not in PREFERRED_ALGORITHMS_BY_LENGTH:
                continue

            if not self._valid_hex(digest):
                continue

            algorithms = PREFERRED_ALGORITHMS_BY_LENGTH[
                len(digest)
            ]

            if (
                self.config.reject_git_commit_patterns
                and self._looks_like_git_context(
                    text,
                    left,
                    right,
                )
            ):
                continue

            source_start = self._map_start(
                left,
                index_map,
            )

            source_end = self._map_end(
                right,
                index_map,
                source_start,
            )

            results.append(
                _Candidate(
                    start=source_start,
                    end=source_end,
                    raw_value=raw_region,
                    explicit_algorithm=None,
                    encoding=HashEncoding.HEX,
                    match_type=(
                        HashMatchType.AMBIGUOUS
                        if len(algorithms) > 1
                        else HashMatchType.EXACT
                    ),
                    priority=70 + len(digest),
                )
            )

            self.metrics.increment("defanged_candidates")

    def _normalize_defanged_digest(
        self,
        value: str,
    ) -> Optional[tuple[str, tuple[NormalizationAction, ...]]]:
        if not value:
            return None

        actions: list[NormalizationAction] = []

        restored = value

        for _ in range(MAX_NORMALIZATION_ROUNDS):
            previous = restored

            replacements = (
                ("[dot]", "."),
                ("(dot)", "."),
                ("{dot}", "."),
                ("[.]", "."),
                ("(.)", "."),
                ("{.}", "."),
            )

            for source, target in replacements:
                if source in restored.lower():
                    restored = re.sub(
                        re.escape(source),
                        target,
                        restored,
                        flags=re.IGNORECASE,
                    )

            if restored != previous:
                if NormalizationAction.DEFANG_RESTORED not in actions:
                    actions.append(
                        NormalizationAction.DEFANG_RESTORED
                    )

            if restored == previous:
                break

        if self.config.remove_internal_ascii_whitespace_from_defanged:
            compact = re.sub(r"[ \t\r\n]+", "", restored)

            if compact != restored:
                restored = compact

                actions.append(
                    NormalizationAction.WHITESPACE_REMOVED
                )

        # A hash is not a dotted textual object after normalization.
        # Dots are separators introduced by defanging, therefore remove
        # them ONLY after confirming that every non-hex character is a
        # known defang separator.
        if "." in restored:
            restored = restored.replace(".", "")

        restored = restored.strip("[](){}<>\"'`")

        if not restored:
            return None

        if len(restored) > DEFAULT_MAX_DIGEST_LENGTH:
            return None

        if not self._valid_hex(restored):
            return None

        return restored.lower(), tuple(actions)

    # ------------------------------------------------------------------
    # Candidate deconfliction
    # ------------------------------------------------------------------

    def _deconflict_candidates(
        self,
        candidates: Sequence[_Candidate],
    ) -> list[_Candidate]:
        if not candidates:
            return []

        ordered = sorted(
            candidates,
            key=lambda candidate: (
                -candidate.priority,
                -(candidate.end - candidate.start),
                candidate.start,
                candidate.end,
            ),
        )

        selected: list[_Candidate] = []

        for candidate in ordered:
            overlaps = False

            for existing in selected:
                if (
                    candidate.start < existing.end
                    and candidate.end > existing.start
                ):
                    overlaps = True
                    break

            if not overlaps:
                selected.append(candidate)

        selected.sort(
            key=lambda candidate: (
                candidate.start,
                candidate.end,
            )
        )

        return selected[: self.config.max_iocs]

    # ------------------------------------------------------------------
    # IOC materialization
    # ------------------------------------------------------------------

    def _materialize_iocs(
        self,
        original_text: str,
        working_text: str,
        candidates: Sequence[_Candidate],
        started: float,
    ) -> list[HashIOC]:
        iocs: list[HashIOC] = []

        for candidate in candidates:
            if self._budget_exceeded(started):
                self.metrics.increment("time_budget_exceeded")
                break

            if len(iocs) >= self.config.max_iocs:
                break

            raw = self._safe_original_slice(
                original_text,
                candidate.start,
                candidate.end,
            )

            normalized = self._normalize_candidate(
                candidate,
                raw,
            )

            if normalized is None:
                continue

            digest = normalized.value
            algorithm = normalized.algorithm

            alternate_algorithms: tuple[HashAlgorithm, ...] = ()

            if algorithm is None:
                possible = PREFERRED_ALGORITHMS_BY_LENGTH.get(
                    len(digest),
                    (),
                )

                if not possible:
                    continue

                algorithm = possible[0]

                if (
                    self.config.include_ambiguous_algorithms
                    and len(possible) > 1
                ):
                    alternate_algorithms = tuple(
                        possible[1:]
                    )

            confidence = self._calculate_confidence(
                candidate=candidate,
                normalized=normalized,
                algorithm=algorithm,
                alternate_algorithms=alternate_algorithms,
            )

            context = self._bounded_context(
                original_text,
                candidate.start,
                candidate.end,
            )

            ioc_id = self._build_ioc_id(
                digest=digest,
                algorithm=algorithm,
                start=candidate.start,
                end=candidate.end,
            )

            metadata = {
                "extractor_version": VERSION,
                "ambiguous": bool(alternate_algorithms),
                "raw_span_length": candidate.end - candidate.start,
            }

            if alternate_algorithms:
                metadata["alternate_algorithms"] = tuple(
                    algorithm.value
                    for algorithm in alternate_algorithms
                )

            iocs.append(
                HashIOC(
                    ioc_id=ioc_id,
                    kind="hash",
                    algorithm=algorithm,
                    value=digest,
                    raw_value=raw,
                    start=candidate.start,
                    end=candidate.end,
                    confidence=confidence,
                    match_type=normalized.match_type,
                    encoding=normalized.encoding,
                    normalization_actions=normalized.actions,
                    alternate_algorithms=alternate_algorithms,
                    source="text",
                    context=context,
                    metadata=metadata,
                )
            )

        return iocs

    def _normalize_candidate(
        self,
        candidate: _Candidate,
        raw: str,
    ) -> Optional[_NormalizedCandidate]:
        actions: list[NormalizationAction] = []

        if candidate.encoding == HashEncoding.SRI_BASE64:
            # Candidate has already been validated and decoded during
            # discovery. Re-decode here to keep materialization independent.
            match = re.search(
                r"(?i)^(sha256|sha384|sha512)-"
                r"([A-Za-z0-9+/]+={0,2})$",
                raw.strip(),
            )

            if not match:
                return None

            try:
                decoded = base64.b64decode(
                    match.group(2),
                    validate=True,
                )
            except (binascii.Error, ValueError):
                return None

            algorithm = {
                "sha256": HashAlgorithm.SHA256,
                "sha384": HashAlgorithm.SHA384,
                "sha512": HashAlgorithm.SHA512,
            }.get(match.group(1).lower())

            if algorithm is None:
                return None

            expected = {
                HashAlgorithm.SHA256: 32,
                HashAlgorithm.SHA384: 48,
                HashAlgorithm.SHA512: 64,
            }[algorithm]

            if len(decoded) != expected:
                return None

            actions.append(
                NormalizationAction.SRI_DECODED
            )

            return _NormalizedCandidate(
                value=decoded.hex(),
                algorithm=algorithm,
                encoding=HashEncoding.SRI_BASE64,
                match_type=HashMatchType.SRI,
                actions=tuple(actions),
            )

        value = raw.strip()

        if value != raw:
            actions.append(
                NormalizationAction.WRAPPER_REMOVED
            )

        explicit_algorithm = candidate.explicit_algorithm

        if explicit_algorithm is not None:
            match_type = HashMatchType.EXACT
        else:
            possible = PREFERRED_ALGORITHMS_BY_LENGTH.get(
                len(value),
                (),
            )

            if not possible:
                return None

            explicit_algorithm = (
                possible[0]
                if len(possible) == 1
                else None
            )

            match_type = (
                HashMatchType.AMBIGUOUS
                if len(possible) > 1
                else HashMatchType.EXACT
            )

        # Defanged normalization.
        if self.config.allow_defanged and any(
            marker in value.lower()
            for marker in (
                "[.]",
                "(.)",
                "{.}",
                "[dot]",
                "(dot)",
                "{dot}",
            )
        ):
            restored = self._normalize_defanged_digest(value)

            if restored is None:
                return None

            value, defang_actions = restored
            actions.extend(defang_actions)

        value = value.lower()

        if explicit_algorithm is not None:
            if not self._valid_digest_for_algorithm(
                value,
                explicit_algorithm,
            ):
                return None
        else:
            if not self._valid_hex(value):
                return None

        return _NormalizedCandidate(
            value=value,
            algorithm=explicit_algorithm,
            encoding=HashEncoding.HEX,
            match_type=match_type,
            actions=self._unique_actions(actions),
        )

    # ------------------------------------------------------------------
    # Deduplication
    # ------------------------------------------------------------------

    def _deduplicate_iocs(
        self,
        iocs: Sequence[HashIOC],
    ) -> list[HashIOC]:
        if not iocs:
            return []

        selected: dict[tuple[str, str], HashIOC] = {}

        for ioc in iocs:
            key = (
                ioc.algorithm.value,
                ioc.value,
            )

            existing = selected.get(key)

            if existing is None:
                selected[key] = ioc
                continue

            # Prefer:
            # 1. explicit exact match
            # 2. SRI
            # 3. higher confidence
            # 4. longer provenance span
            current_score = (
                self._match_priority(ioc.match_type),
                ioc.confidence,
                ioc.end - ioc.start,
            )

            existing_score = (
                self._match_priority(existing.match_type),
                existing.confidence,
                existing.end - existing.start,
            )

            if current_score > existing_score:
                selected[key] = ioc

        result = sorted(
            selected.values(),
            key=lambda item: (
                item.start,
                item.end,
                item.algorithm.value,
                item.value,
            ),
        )

        return result[: self.config.max_iocs]

    @staticmethod
    def _match_priority(match_type: HashMatchType) -> int:
        return {
            HashMatchType.SRI: 3,
            HashMatchType.EXACT: 2,
            HashMatchType.AMBIGUOUS: 1,
        }.get(match_type, 0)

    # ------------------------------------------------------------------
    # Validation
    # ------------------------------------------------------------------

    @staticmethod
    def _valid_hex(value: str) -> bool:
        if not value:
            return False

        length = len(value)

        if length not in {
            32,
            40,
            56,
            64,
            96,
            128,
        }:
            return False

        # translate() avoids allocating one transformed character at
        # a time and gives a predictable O(n) validation.
        return (
            value.translate(
                str.maketrans(
                    "",
                    "",
                    "0123456789abcdefABCDEF",
                )
            )
            == ""
        )

    @staticmethod
    def _valid_digest_for_algorithm(
        digest: str,
        algorithm: HashAlgorithm,
    ) -> bool:
        info = ALGORITHM_INFO.get(algorithm)

        if info is None:
            return False

        if len(digest) != info.hex_length:
            return False

        return HashIOCExtractor._valid_hex_exact_length(
            digest,
            info.hex_length,
        )

    @staticmethod
    def _valid_hex_exact_length(
        value: str,
        expected_length: int,
    ) -> bool:
        if len(value) != expected_length:
            return False

        for char in value:
            if char not in "0123456789abcdefABCDEF":
                return False

        return True

    def _looks_like_non_hash_token(
        self,
        text: str,
        start: int,
        end: int,
    ) -> bool:
        """
        Conservative rejection of obvious machine-generated identifiers.

        This must not become too aggressive because many legitimate malware
        hashes occur near arbitrary text.
        """

        value = text[start:end]

        # Very long repeated single-character values are almost certainly
        # test/padding data rather than cryptographic digests.
        if len(set(value.lower())) <= 2:
            return True

        return False

    def _looks_like_git_context(
        self,
        text: str,
        start: int,
        end: int,
    ) -> bool:
        radius = 48

        left = max(0, start - radius)
        right = min(len(text), end + radius)

        context = text[left:right].lower()

        git_terms = (
            "git commit",
            "commit:",
            "commit ",
            "revision:",
            "revision ",
            "git-sha",
            "git_sha",
            "changeset:",
            "changeset ",
            "tree:",
            "blob:",
        )

        return any(term in context for term in git_terms)

    # ------------------------------------------------------------------
    # Confidence
    # ------------------------------------------------------------------

    def _calculate_confidence(
        self,
        *,
        candidate: _Candidate,
        normalized: _NormalizedCandidate,
        algorithm: HashAlgorithm,
        alternate_algorithms: Sequence[HashAlgorithm],
    ) -> float:
        score = 0.70

        if candidate.explicit_algorithm is not None:
            score += 0.15

        if normalized.encoding == HashEncoding.SRI_BASE64:
            score += 0.10

        if (
            normalized.match_type == HashMatchType.AMBIGUOUS
            or alternate_algorithms
        ):
            score -= 0.10

        # Algorithm quality affects identity confidence, not maliciousness.
        if algorithm in {
            HashAlgorithm.SHA256,
            HashAlgorithm.SHA384,
            HashAlgorithm.SHA512,
            HashAlgorithm.SHA3_256,
            HashAlgorithm.SHA3_384,
            HashAlgorithm.SHA3_512,
            HashAlgorithm.BLAKE2S,
            HashAlgorithm.BLAKE2B,
        }:
            score += 0.03

        if algorithm in {
            HashAlgorithm.MD5,
            HashAlgorithm.SHA1,
        }:
            score -= 0.03

        if candidate.raw_value != normalized.value:
            score -= 0.03

        return max(0.0, min(1.0, score))

    # ------------------------------------------------------------------
    # Span mapping
    # ------------------------------------------------------------------

    @staticmethod
    def _map_start(
        working_index: int,
        index_map: Sequence[int],
    ) -> int:
        if not index_map:
            return 0

        if working_index < 0:
            return index_map[0]

        if working_index >= len(index_map):
            return index_map[-1]

        return index_map[working_index]

    @staticmethod
    def _map_end(
        working_index: int,
        index_map: Sequence[int],
        source_start: int,
    ) -> int:
        if not index_map:
            return source_start + 1

        last_working_index = max(
            0,
            min(
                len(index_map) - 1,
                working_index - 1,
            ),
        )

        return max(
            source_start + 1,
            index_map[last_working_index] + 1,
        )

    @staticmethod
    def _safe_slice(
        index_map: Sequence[int],
        start: int,
        end: int,
        *,
        fallback: str,
    ) -> str:
        if not index_map:
            return fallback

        source_start = index_map[
            max(0, min(len(index_map) - 1, start))
        ]

        source_last = index_map[
            max(0, min(len(index_map) - 1, end - 1))
        ]

        if source_last < source_start:
            return fallback

        return fallback

    @staticmethod
    def _safe_original_slice(
        text: str,
        start: int,
        end: int,
    ) -> str:
        start = max(0, min(len(text), start))
        end = max(start, min(len(text), end))

        return text[start:end]

    # ------------------------------------------------------------------
    # Context
    # ------------------------------------------------------------------

    def _bounded_context(
        self,
        text: str,
        start: int,
        end: int,
    ) -> str:
        radius = self.config.max_context_chars // 2

        left = max(0, start - radius)
        right = min(len(text), end + radius)

        context = text[left:right]

        # Remove dangerous control characters while preserving useful
        # whitespace.
        cleaned_chars: list[str] = []

        for char in context:
            category = unicodedata.category(char)

            if category.startswith("C") and char not in "\n\t\r":
                continue

            cleaned_chars.append(char)

        cleaned = "".join(cleaned_chars)

        if len(cleaned) > self.config.max_context_chars:
            cleaned = cleaned[
                : self.config.max_context_chars
            ]

        return cleaned

    # ------------------------------------------------------------------
    # IOC identity
    # ------------------------------------------------------------------

    @staticmethod
    def _build_ioc_id(
        *,
        digest: str,
        algorithm: HashAlgorithm,
        start: int,
        end: int,
    ) -> str:
        """
        Stable identity for one source occurrence.

        Digest + algorithm identifies the IOC.
        Span is included to preserve distinct occurrences in forensic
        extraction. Reputation caches should normally use algorithm+digest
        instead of this occurrence ID.
        """

        material = (
            f"hash|{algorithm.value}|{digest}|"
            f"{start}:{end}"
        )

        return sha256(
            material.encode("utf-8")
        ).hexdigest()

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------

    @staticmethod
    def _unique_actions(
        actions: Iterable[NormalizationAction],
    ) -> tuple[NormalizationAction, ...]:
        seen: set[NormalizationAction] = set()
        result: list[NormalizationAction] = []

        for action in actions:
            if action not in seen:
                seen.add(action)
                result.append(action)

        return tuple(result)

    @staticmethod
    def _budget_exceeded(started: float) -> bool:
        # This helper is intentionally only used where the caller already
        # knows the configured budget. Actual checks happen through the
        # instance method below.
        return False

    def _budget_exceeded_instance(
        self,
        started: float,
    ) -> bool:
        return (
            time.monotonic() - started
            >= self.config.max_extraction_seconds
        )

    def _elapsed_ms(self, started: float) -> float:
        return max(
            0.0,
            (time.monotonic() - started) * 1000.0,
        )

    def _failure(
        self,
        started: float,
        *,
        errors: tuple[str, ...],
    ) -> HashExtractionResult:
        return HashExtractionResult(
            success=False,
            iocs=(),
            truncated=False,
            elapsed_ms=self._elapsed_ms(started),
            candidates_seen=0,
            extractor_errors=errors,
            metrics_snapshot=self.metrics.snapshot(),
        )

_DEFAULT_EXTRACTOR: Optional[HashIOCExtractor] = None
_DEFAULT_EXTRACTOR_LOCK = threading.Lock()


def get_default_hash_extractor() -> HashIOCExtractor:
    global _DEFAULT_EXTRACTOR

    if _DEFAULT_EXTRACTOR is None:
        with _DEFAULT_EXTRACTOR_LOCK:
            if _DEFAULT_EXTRACTOR is None:
                _DEFAULT_EXTRACTOR = HashIOCExtractor()

    return _DEFAULT_EXTRACTOR


def extract_hash_iocs(
    text: str,
    *,
    config: Optional[HashExtractorConfig] = None,
) -> HashExtractionResult:
    """
    Convenience synchronous API.

    Uses the singleton only when no custom configuration is supplied.
    """

    if config is None:
        return get_default_hash_extractor().extract(text)

    return HashIOCExtractor(config).extract(text)


async def extract_hash_iocs_async(
    text: str,
    *,
    config: Optional[HashExtractorConfig] = None,
) -> HashExtractionResult:
    """
    Convenience asynchronous API.
    """

    if config is None:
        return await get_default_hash_extractor().extract_async(text)

    return await HashIOCExtractor(config).extract_async(text)


# ============================================================================
# Digest utility API
# ============================================================================


def canonicalize_digest(
    value: str,
    *,
    algorithm: Optional[HashAlgorithm] = None,
) -> Optional[str]:
    """
    Canonicalize an explicitly supplied hexadecimal digest.

    This function does not perform reputation lookup.
    """

    if not isinstance(value, str):
        return None

    candidate = value.strip().lower()

    if algorithm is not None:
        if not HashIOCExtractor._valid_digest_for_algorithm(
            candidate,
            algorithm,
        ):
            return None

        return candidate

    if not HashIOCExtractor._valid_hex(candidate):
        return None

    return candidate


def infer_hash_algorithms(
    digest: str,
) -> tuple[HashAlgorithm, ...]:
    """
    Return all algorithms compatible with the digest length.

    Example:
        64 hex characters ->
        SHA-256, SHA3-256, BLAKE2S
    """

    if not isinstance(digest, str):
        return ()

    candidate = digest.strip().lower()

    if not HashIOCExtractor._valid_hex(candidate):
        return ()

    return PREFERRED_ALGORITHMS_BY_LENGTH.get(
        len(candidate),
        (),
    )
